**Goal:** Establish GUT 9.7.1 as an exact-head formal test authority with real finite-puzzle consumers, JUnit evidence, pinned-vendor reconciliation, and a production-mutation guard while retaining the legacy custom runner during migration.

**Architecture:** Add a parallel `GUT 9.7.1 Tests` workflow rather than replacing `Godot Tests`. from tools.gut_phase_b_guard import compare_vendor, snapshot


```python
def test_vendor_compare_allows_only_redundant_load_steps(tmp_path):
    official = tmp_path / "official"
    local = tmp_path / "local"
    official.mkdir(); local.mkdir()
    (official / "GutScene.tscn").write_text('[gd_scene load_steps=4 format=3]\n[node name="A"]\n')
    (local / "GutScene.tscn").write_text('[gd_scene format=3]\n[node name="A"]\n')
    report = compare_vendor(local, official)
    assert report["source_divergence"] == []
    assert report["normalized_scene_metadata"] == ["GutScene.tscn"]


def test_vendor_compare_rejects_source_change(tmp_path):
    official = tmp_path / "official"; local = tmp_path / "local"
    official.mkdir(); local.mkdir()
    (official / "gut.gd").write_text("extends Node\n")
    (local / "gut.gd").write_text("extends RefCounted\n")
    assert compare_vendor(local, official)["source_divergence"] == ["gut.gd"]


def test_snapshot_hashes_protected_files_only(tmp_path):
    (tmp_path / "project.godot").write_text("config")
    (tmp_path / "game").mkdir(); (tmp_path / "game" / "core.gd").write_text("code")
    result = snapshot(tmp_path, ["project.godot", "game"])
    assert sorted(result) == ["game/core.gd", "project.godot"]
```
